### 优化建议

```c
// sheet_writer.c优化：单次扫描
static cxlsb_error_t write_sheet_data_optimized(...) {
    for (int row = start_row; row < start_row + row_count; row++) {
        cxlsb_cell_data_t cells[col_count];  // 缓存一行数据
        int first_col = -1, last_col = -1;
        
        for (int col = 0; col < col_count; col++) {
            cells[col] = supplier(row, col, user_data);  // 单次调用
            if (cells[col].type != CXLSB_CELL_BLANK) {
                if (first_col < 0) first_col = col;
                last_col = col;
            }
        }
        
        if (first_col >= 0) {
            cxlsb_biff12_writer_write_row_header(...);
            for (int col = first_col; col <= last_col; col++) {
                // 直接使用cells[col]，不再调用supplier
            }
        }
    }
}
```
